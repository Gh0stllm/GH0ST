type entry struct {
	Offset int64
	Size   uint32
}
type frozenDB struct {
	ethdb.KeyValueStore
	file  *os.File
	index map[common.Hash]entry
	reads uint64
}

func (d *frozenDB) Get(key []byte) ([]byte, error) {
	if bytes.HasPrefix(key, rawdb.CodePrefix) && len(key) == len(rawdb.CodePrefix)+32 {
		key = key[len(rawdb.CodePrefix):]
	}
	if len(key) != 32 {
		return nil, fmt.Errorf("non-preimage lookup %x", key)
	}
	e, ok := d.index[common.BytesToHash(key)]
	if !ok {
		return nil, errors.New("frozen preimage absent")
	}
	b := make([]byte, e.Size)
	_, err := d.file.ReadAt(b, e.Offset)
	d.reads++
	return b, err
}
func (d *frozenDB) Has(key []byte) (bool, error) { _, e := d.Get(key); return e == nil, nil }
func (d *frozenDB) Put([]byte, []byte) error     { return errors.New("study is read only") }
func (d *frozenDB) Delete([]byte) error          { return errors.New("study is read only") }
func (d *frozenDB) Close() error                 { return nil }

type global struct {
	Name  string
	Value uint64
}
type meta struct {
	Version          uint32
	Module           [32]byte
	Root             [32]byte
	MemoryBytes      uint32
	Globals          []global
	Generation       uint64
	Head             [32]byte
	Deltas           uint32
	History, Request string
	Progress         json.RawMessage
}
type counter struct{ N uint64 }

func (c *counter) Write(b []byte) (int, error) { c.N += uint64(len(b)); return len(b), nil }
func must(err error) {
	if err != nil {
		panic(err)
	}
}
func check(ok bool, msg string) {
	if !ok {
		panic(msg)
	}
}
func hash(b []byte) string { x := sha256.Sum256(b); return hex.EncodeToString(x[:]) }

func namespace(owner common.Address, job [32]byte) []byte {
	return crypto.Keccak256([]byte("GL3_CHESS_SLICE_JOB_V1"), owner[:], job[:])
}
func pageAddress(ns []byte, i int) common.Address {
	var n [8]byte
	binary.BigEndian.PutUint64(n[:], uint64(i))
	return common.BytesToAddress(crypto.Keccak256([]byte("GL3_CHESS_SLICE_CELL_V1"), ns, n[:])[12:])
}
func deltaAddress(h [32]byte, i int) common.Address {
	var n [4]byte
	binary.LittleEndian.PutUint32(n[:], uint32(i))
	return common.BytesToAddress(crypto.Keccak256([]byte("GL3_CHESS_IMMUTABLE_DELTA_V1"), h[:], n[:])[12:])
}
func metadata(db *state.StateDB, ns []byte) meta {
	s := storage.NewGeth(db, burn.NewSystemBurner(nil, false)).OpenSubStorage([]byte("GL3_CHESS_SLICE_META_V1")).OpenSubStorage(ns)
	b, e := s.GetBytes()
	must(e)
	var m meta
	must(json.Unmarshal(b, &m))
	check(m.Version == 2, "metadata version")
	return m
}

// Exact existing Tree.NewTree/Root commitment, independently reproduced for byte checks.
func imageRoot(memory []byte, globals []global, module [32]byte) [32]byte {
	n := 1
	for n < (768*1024*1024+4095)/4096 {
		n *= 2
	}
	tree := make([][32]byte, n*2)
	for i := 0; i < len(memory)/4096; i++ {
		h := sha256.New()
		h.Write([]byte("GL3_CHESS_VM_PAGE_V1"))
		var b [8]byte
		binary.LittleEndian.PutUint64(b[:], uint64(i))
		h.Write(b[:])
		h.Write(memory[i*4096 : (i+1)*4096])
		copy(tree[n+i][:], h.Sum(nil))
	}
	for i := n - 1; i > 0; i-- {
		var b [65]byte
		b[0] = 1
		copy(b[1:33], tree[2*i][:])
		copy(b[33:], tree[2*i+1][:])
		tree[i] = sha256.Sum256(b[:])
	}
	h := sha256.New()
	h.Write([]byte("GL3_CHESS_IMAGE_V1"))
	h.Write(module[:])
	h.Write(tree[1][:])
	var b [8]byte
	binary.LittleEndian.PutUint64(b[:], uint64(len(memory)))
	h.Write(b[:])
	for _, g := range globals {
		h.Write([]byte(g.Name))
		h.Write([]byte{0})
		binary.LittleEndian.PutUint64(b[:], g.Value)
		h.Write(b[:])
	}
	var result [32]byte
	copy(result[:], h.Sum(nil))
	return result
}

func main() {
	started := time.Now()
	runtime.GOMAXPROCS(2)
	if len(os.Args) != 2 {
		panic("output report path required")
	}
	outDir := os.Args[1]
	f, e := os.Open("/root/bounded-restore/preimages.bin")
	must(e)
	defer f.Close()
	d := &frozenDB{file: f, index: make(map[common.Hash]entry, 410000)}
	var pos int64
	var records int
	sha := sha256.New()
	for {
		var n [8]byte
		_, err := io.ReadFull(f, n[:])
		if err == io.EOF {
			break
		}
		must(err)
		sha.Write(n[:])
		size := binary.LittleEndian.Uint64(n[:])
		check(size > 0 && size <= 64*1024*1024, "preimage length")
		b := make([]byte, int(size))
		_, err = io.ReadFull(f, b)
		must(err)
		sha.Write(b)
		d.index[crypto.Keccak256Hash(b)] = entry{pos + 8, uint32(size)}
		pos += 8 + int64(size)
		records++
	}
	headerBytes, e := d.Get(common.HexToHash("0x4b0cb7e808fd0e32ea667114adf94e67fd7c6617a545e7fb473aab0afb4eb0f0").Bytes())
	must(e)
	var header types.Header
	must(rlp.DecodeBytes(headerBytes, &header))
	raw := rawdb.NewDatabase(d)
	db, e := state.NewDeterministic(header.Root, state.NewDatabase(triedb.NewDatabase(raw, nil), nil))
	must(e)
	owner := common.HexToAddress("0x26E554a8acF9003b83495c7f45F06edCB803d4e3")
	var job [32]byte
	job[0] = 0x33
	seedNS := crypto.Keccak256([]byte("GL3_CHESS_SLICE_SEED_V1"))
	seed := metadata(db, seedNS)
	m := metadata(db, namespace(owner, job))
	check(m.MemoryBytes >= seed.MemoryBytes && m.MemoryBytes%65536 == 0, "memory lengths")
	base := make([]byte, m.MemoryBytes)
	seedAccounts := 0
	for i := 0; i < int(seed.MemoryBytes)/4096; i++ {
		code := db.GetCodeUncached(pageAddress(seedNS, i))
		if len(code) == 0 {
			continue
		}
		check(bytes.HasPrefix(code, []byte{0, 'G', 'L', '3', 'S', 1}) && (len(code) == 6 || len(code) == 4102), "seed page encoding")
		copy(base[i*4096:], code[6:])
		seedAccounts++
	}
	check(imageRoot(base[:seed.MemoryBytes], seed.Globals, seed.Module) == seed.Root, "genesis image commitment mismatch")
	memory := bytes.Clone(base)
	seen := make([]byte, (len(memory)/32+7)/8)
	head := m.Head
	var rawDeltas, codeBytes uint64
	var deltaAccounts, nonzeroNonces, nonzeroBalances int
	for left := m.Deltas; left > 0; left-- {
		first := db.GetCodeUncached(deltaAddress(head, 0))
		check(len(first) >= 10 && bytes.Equal(first[:6], []byte{0, 'G', 'L', '3', 'D', 1}), "delta first chunk")
		n := int(binary.LittleEndian.Uint32(first[6:10]))
		check(n >= 44 && n <= 32*1024*1024, "delta size")
		body := make([]byte, n)
		for at := 0; at < n; at += 16384 {
			end := at + 16384
			if end > n {
				end = n
			}
			a := deltaAddress(head, at/16384)
			code := first
			if at > 0 {
				code = db.GetCodeUncached(a)
			}
			check(len(code) == 10+end-at && bytes.Equal(code[:6], first[:6]) && int(binary.LittleEndian.Uint32(code[6:10])) == n, "delta chunk")
			copy(body[at:end], code[10:])
			deltaAccounts++
			codeBytes += uint64(len(code))
			if db.GetNonce(a) != 0 {
				nonzeroNonces++
			}
			if !db.GetBalance(a).IsZero() {
				nonzeroBalances++
			}
		}
		check(sha256.Sum256(body) == head && string(body[:4]) == "GSD1", "delta digest")
		size := int(binary.LittleEndian.Uint32(body[36:40]))
		count := int(binary.LittleEndian.Uint32(body[40:44]))
		check(size <= len(memory) && size%65536 == 0 && 44+36*count == len(body), "delta header")
		priorOffset := -1
		for at := 44; at < len(body); at += 36 {
			offset := int(binary.LittleEndian.Uint32(body[at : at+4]))
			check(offset%32 == 0 && offset > priorOffset && offset+32 <= size, "delta word order")
			priorOffset = offset
			word := offset / 32
			mask := byte(1 << uint(word%8))
			if seen[word/8]&mask == 0 {
				copy(memory[offset:offset+32], body[at+4:at+36])
				seen[word/8] |= mask
			}
		}
		copy(head[:], body[4:36])
		rawDeltas += uint64(n)
	}
	check(head == ([32]byte{}), "delta ancestry not exhausted")
	must(db.Error())
	gotRoot := imageRoot(memory, m.Globals, m.Module)
	check(gotRoot == m.Root, "authoritative preterminal image root mismatch")

	check(m.Module == seed.Module && hex.EncodeToString(m.Module[:]) == "1d650410655d93624801a0fa45d64d72ec5d209115672b5848f010037da5e5bc", "module identity")
	must(os.MkdirAll(outDir, 0700))
	must(os.WriteFile(outDir+"/seed.bin", base[:seed.MemoryBytes], 0600))
	must(os.WriteFile(outDir+"/image.bin", memory, 0600))
	result := map[string]any{"SeedGlobals": seed.Globals, "ImageGlobals": m.Globals, "ImageRoot": hex.EncodeToString(m.Root[:]), "ImageSHA256": hash(memory), "Completed": false, "Scope": "Authenticated PRETERMINAL first-decision state only, no engine/VM/block execution", "PrestateBlockNumber": header.Number.Uint64(), "PrestateStateRoot": header.Root, "Deltas": m.Deltas, "MemoryBytes": len(memory), "SeedMemoryBytes": seed.MemoryBytes, "ModuleSHA256": hex.EncodeToString(m.Module[:]), "PreimagesSHA256": hex.EncodeToString(sha.Sum(nil)), "Seconds": time.Since(started).Seconds(), "allExecutionOnRunpod": true, "engineExecuted": false}
	out, e := json.MarshalIndent(result, "", "  ")
	must(e)
	must(os.WriteFile(outDir+"/fixture.json", append(out, '\n'), 0600))
	fmt.Println(string(out))
}

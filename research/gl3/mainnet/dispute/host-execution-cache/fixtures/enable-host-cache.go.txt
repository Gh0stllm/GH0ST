// After each builder's Build(t), before challenge actors admit native work:
clients := gl3EnableHostExecutionCache(t, b.L2.ConsensusNode.StatelessBlockValidator)
// Use the same hook for an independent evil builder if it owns a BOLD client.
// Its heavy calls must still delegate to the one existing honest provider.

func gl3EnableHostExecutionCache(t *testing.T, v *staker.StatelessBlockValidator) []*client.BOLDExecutionClient {
    t.Helper()
    if v == nil { t.Fatal("stateless validator missing") }
    var clients []*client.BOLDExecutionClient
    for _, spawner := range v.BOLDExecutionSpawners() {
        cached, ok := spawner.(*client.BOLDExecutionClient)
        if !ok { t.Fatal("unexpected BOLD execution spawner type") }
        // Existing StatelessBlockValidator.Start starts the underlying clients
        // and BOLD clients before this hook; Enable fails closed otherwise.
        if err := cached.EnableExperimentalRunCache(client.ExperimentalRunCacheConfig{
            MaxEntries: 1, IdleTimeout: 4 * time.Hour,
        }); err != nil { t.Fatal(err) }
        clients = append(clients, cached)
    }
    if len(clients) == 0 { t.Fatal("no cached BOLD client enabled") }
    return clients
}
